```ts
import { Bot } from "grammy";
import {
  makeStartHandler,
  makeChapterCallbackHandler,
  makeConsentCallbackHandler,
} from "./src/handlers/start.js";
import { makeHelpHandler } from "./src/handlers/help.js";
import { createDbClient } from "./src/db/client.js";

const { db } = createDbClient(process.env.DATABASE_URL!);

const bot = new Bot("000000:DUMMY_TOKEN_NOT_REAL", {
  botInfo: {
    id: 1,
    is_bot: true,
    first_name: "Test",
    username: "test_bot",
    can_join_groups: true,
    can_read_all_group_messages: false,
    supports_inline_queries: false,
  },
});

const captured: Array<{ method: string; payload: unknown }> = [];
bot.api.config.use(async (_prev, method, payload) => {
  captured.push({ method, payload });
  return { ok: true, result: true } as never;
});

bot.command("start", makeStartHandler(db));
bot.command("help", makeHelpHandler(db));
bot.callbackQuery(/^chapter:(.+)$/, makeChapterCallbackHandler(db));
bot.callbackQuery("consent:agree", makeConsentCallbackHandler(db));

function messageUpdate(updateId: number, tgId: number, text: string, username?: string) {
  return {
    update_id: updateId,
    message: {
      message_id: updateId,
      date: Math.floor(Date.now() / 1000),
      chat: { id: tgId, type: "private" as const },
      from: { id: tgId, is_bot: false, first_name: "Test", ...(username ? { username } : {}) },
      text,
      entities: [{ type: "bot_command" as const, offset: 0, length: text.split(" ")[0].length }],
    },
  };
}

function callbackUpdate(updateId: number, tgId: number, data: string, callbackQueryId: string) {
  return {
    update_id: updateId,
    callback_query: {
      id: callbackQueryId,
      from: { id: tgId, is_bot: false, first_name: "Test" },
      message: {
        message_id: updateId - 1,
        date: Math.floor(Date.now() / 1000),
        chat: { id: tgId, type: "private" as const },
      },
      data,
      chat_instance: "1",
    },
  };
}

async function main() {
  await bot.init();
  // --- dispatch calls inserted per-case below, in order ---
  console.log(JSON.stringify(captured, null, 2));
  process.exit(0);
}

main();
```

## ORDERING CASE — consent strictly before any chapter write (not a literal AC, but

- Fixture: `DELETE FROM chapters;` then insert exactly one active chapter (reuse the
  AC2a fixture shape with a fresh code, e.g. `ord1`).
- Dispatch, with a DB read-back interleaved **between each step** (not only at the

     Read-back: `SELECT chapter_id, consent_pd_at FROM users WHERE tg_id = 700000007;`
     Expected: `chapter_id IS NULL`, `consent_pd_at IS NULL` — the consent prompt was
     sent and NOTHING chapter-related has been queried or written yet.
