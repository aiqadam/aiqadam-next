## AC2 — `messages/kk.json` key-set matches `messages/ru.json` exactly, all values non-empty

- Route: N/A (data file check, script-based)
- Viewport: N/A
- Theme: N/A
- Steps: from the repo root, run this Node script (exact comparison — key-set equality
  by recursive leaf-path collection, plus an emptiness check):
  ```bash
  node -e "
  function leaves(obj, prefix) {
    let out = [];
    for (const k of Object.keys(obj)) {
      const p = prefix ? prefix + '.' + k : k;
      if (obj[k] && typeof obj[k] === 'object' && !Array.isArray(obj[k])) {
        out = out.concat(leaves(obj[k], p));
      } else {
        out.push(p);
      }
    }
    return out;
  }
  const ru = require('./messages/ru.json');
  const kk = require('./messages/kk.json');
  const ruKeys = leaves(ru, '').sort();
  const kkKeys = leaves(kk, '').sort();
  const missing = ruKeys.filter(k => !kkKeys.includes(k));
  const extra = kkKeys.filter(k => !ruKeys.includes(k));
  function getVal(obj, path) { return path.split('.').reduce((o,p)=>o&&o[p], obj); }
  const empty = kkKeys.filter(k => { const v = getVal(kk,k); return v === '' || v === null || v === undefined; });
  console.log('ru_count=' + ruKeys.length);
  console.log('kk_count=' + kkKeys.length);
  console.log('missing=' + JSON.stringify(missing));
  console.log('extra=' + JSON.stringify(extra));
  console.log('empty=' + JSON.stringify(empty));
  "
  ```
- Expected output, exactly (**actually run today against the real files — this is the
  real observed output, not a projection**):
  ```
  ru_count=177
  kk_count=177
  missing=[]
  extra=[]
  empty=[]
  ```

## AC2b — every `kk.json` value is the verbatim `ru.json` value, including `Metadata.ogLocale`

Unlike REQ-003 (whose `en.json` had one deliberate divergence at `Metadata.ogLocale =
"en_US"`), REQ-005's requirement text states `Metadata.ogLocale` is **also** verbatim
`"ru_RU"` for this locale — the real `kk_KZ`-style value is explicitly deferred to
REQ-006. So this case expects **zero** differing keys, not one.

- Route: N/A (data file check, script-based)
- Viewport: N/A
- Theme: N/A
- Steps: from the repo root, run:
  ```bash
  node -e "
  function leaves(obj, prefix) {
    let out = [];
    for (const k of Object.keys(obj)) {
      const p = prefix ? prefix + '.' + k : k;
      if (obj[k] && typeof obj[k] === 'object' && !Array.isArray(obj[k])) {
        out = out.concat(leaves(obj[k], p));
      } else {
        out.push(p);
      }
    }
    return out;
  }
  const ru = require('./messages/ru.json');
  const kk = require('./messages/kk.json');
  function getVal(obj, path) { return path.split('.').reduce((o,p)=>o&&o[p], obj); }
  const diffs = leaves(ru,'').filter(k => getVal(ru,k) !== getVal(kk,k));
  console.log('diff_keys=' + JSON.stringify(diffs));
  console.log('kk_ogLocale=' + JSON.stringify(getVal(kk, 'Metadata.ogLocale')));
  console.log('ru_ogLocale=' + JSON.stringify(getVal(ru, 'Metadata.ogLocale')));
  "
  ```
- Expected output, exactly (**actually run today — real observed output**):
  ```
  diff_keys=[]
  kk_ogLocale="ru_RU"
  ru_ogLocale="ru_RU"
  ```
  `diff_keys=[]` is the critical assertion: it means all 177 leaf values in `kk.json`,
  including `Metadata.ogLocale`, are character-for-character identical to `ru.json`. A
  non-empty `diff_keys` (e.g. `["Metadata.ogLocale"]`, the REQ-003 shape) would mean
  `kk.json` was built by copy-adapting REQ-003's pattern instead of following REQ-005's
  own explicit no-divergence instruction — this case is what would catch that mistake.
