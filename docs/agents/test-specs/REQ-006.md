## CFP verbatim-domain-term check (cross-reference for AC1's stated exception)

The requirement text names CFP as "the sole genuine verbatim-English domain term" — this
is checked directly rather than only asserted, since it's the boundary condition AC1's
34-key exception list depends on:

```
node -e "
const kk = require('./messages/kk.json');
function leaves(obj, prefix) {
  let out = [];
  for (const k of Object.keys(obj)) {
    const p = prefix ? prefix + '.' + k : k;
    if (obj[k] && typeof obj[k] === 'object' && !Array.isArray(obj[k])) out = out.concat(leaves(obj[k], p));
    else out.push([p, obj[k]]);
  }
  return out;
}
const all = leaves(kk, '');
const terms = ['Event','Chapter','Talk','Speaker'];
for (const t of terms) {
  const re = new RegExp('\\\\b' + t + 's?\\\\b', 'g');
  let count = 0;
  const hits = [];
  for (const [k,v] of all) { const m = String(v).match(re); if (m) { count += m.length; hits.push(k); } }
  console.log(t + '_count=' + count, JSON.stringify(hits));
}
console.log('CFP_count=' + (JSON.stringify(all).match(/CFP/g)||[]).length);
"
```

```
Chapter_count=0 []
Talk_count=0 []
Speaker_count=0 []
CFP_count=2
```
